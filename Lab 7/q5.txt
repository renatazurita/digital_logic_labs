module main(input [15:0] sw,
            output [7:0] D0_seg, D1_seg,
            output[3:0] D0_a, D1_a);

    assign D0_a = 4'b1110; //only turn on rightmost led on left display
    assign D1_a = 4'b1110; //only turn on rightmost led on right display

    wire [3:0] negate;
    wire [7:0] positiveNum, negativeNum;
    assign D0_seg = (sw[3] == 0) ? 8'b11111111 : 8'b10111111; //outputs negative sign if switch3 is on
    assign D1_seg = (sw[3] == 0) ? positiveNum : negativeNum; //chooses whether to output positive hexEncode or negative hexEncode

    negate n0(sw[3:0], negate); //perform two's complement on switch input
    hexEncode h0(sw[3:0],positiveNum); //ouput unsigned operation
    hexEncode h1(negate, negativeNum); //output signed operation



endmodule

module negate(input [3:0] in,
              output [3:0] out);
    assign out = ~in + 1;

endmodule

module hexEncode(input [3:0] binary,
                 output [7:0] hexadecimal);
    reg [7:0] hex1;
    always @(*) begin
        case (binary)
            4'b0000 : hex1 = 8'b11000000;
            4'b0001 : hex1 = 8'b11111001;
            4'b0010 : hex1 = 8'b10100100;
            4'b0011 : hex1 = 8'b10110000;
            4'b0100 : hex1 = 8'b10011001;
            4'b0101 : hex1 = 8'b10010010;
            4'b0110 : hex1 = 8'b10000010;
            4'b0111 : hex1 = 8'b11111000;
            4'b1000 : hex1 = 8'b10000000;
            4'b1001 : hex1 = 8'b10010000;
            4'b1010 : hex1 = 8'b10001000;
            4'b1011 : hex1 = 8'b10000011;
            4'b1100 : hex1 = 8'b11000110;
            4'b1101 : hex1 = 8'b10100001;
            4'b1110 : hex1 = 8'b10000110;
            4'b1111 : hex1 = 8'b10001110;
        endcase
    end
    assign hexadecimal = hex1;

endmodule
