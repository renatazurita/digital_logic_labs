module main(input mclk, input [15:0] sw, output [7:0] D0_seg, output [3:0] D0_a);
  wire clkDiv;
  wire [7:0] out1, out2, out3, out4;
  reg [1:0] count;
  reg [7:0] num_out;
  reg [3:0] light;

  counter c1(mclk, clkDiv);
  hexEncode h1(sw[3:0], out1);
  hexEncode h2(sw[7:4], out2);
  hexEncode h3(sw[11:8], out3);
  hexEncode h4(sw[15:12], out4);

  always @(posedge clkDiv) begin
    count <= count + 1;

    num_out <= (count == 0) ? out1 :
               (count == 1) ? out2 :
               (count == 2) ? out3 : out4;
    light   <= (count == 0) ? 4'b1110 :
               (count == 1) ? 4'b1101 :
               (count == 2) ? 4'b1011 : 4'b0111;

    if (count == 3)
      count <= 0;
  end

  assign D0_a = light;
  assign D0_seg = num_out;

endmodule

module counter(input clk, output clkDiv);
  reg [25:0] count;
  reg temp;
  localparam maxCount = (50000 - 1);

  always @(posedge clk) begin
    if (count == maxCount) begin
      temp <= ~temp;
      count <= 0;
    end
    else
      count <= count + 1;
  end

  assign clkDiv = temp;

endmodule

module hexEncode(input [3:0] bin, output [7:0] hex);
  wire a, b, c, d;

  assign a = bin[3];
  assign b = bin[2];
  assign c = bin[1];
  assign d = bin[0];

  assign hex[0] = (~a&~b&~c&d) | (~a&b&~c&~d) | (a&b&~c&d) | (a&~b&c&d);
  assign hex[1] = (~a&b&~c&d) | (~a&b&c&~d) | (a&~b&c&d) | (a&b&~c&~d) | (a&b&c);
  assign hex[2] = (~a&~b&c&~d) | (a&b&~c&~d) | (a&b&c);
  assign hex[3] = (~a&~b&~c&d) | (~a&b&~c&~d) | (~a&b&c&d) | (a&~b&c&~d) | (a&b&c&d);
  assign hex[4] = (~a&~b&~c&d) | (~a&~b&c&d) | (~a&b&~c&~d) | (~a&b&~c&d) | (~a&b&c&d) | (a&~b&~c&d);
  assign hex[5] = (~a&~b&~c&d) | (~a&~b&c&~d) | (~a&~b&c&d) | (~a&b&c&d) | (a&b&~c&d);
  assign hex[6] = (~a&~b&~c&~d) | (~a&~b&~c&d) | (~a&b&c&d) | (a&b&~c&~d);
  assign hex[7] = 1;

endmodule
